cmake_minimum_required(VERSION 3.21)
project(ContainerShowcase LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SHOWCASE_PACK_DIR "" CACHE PATH "The verified shared portable pack")
set(SHOWCASE_FONT_DIR "" CACHE PATH "The reviewed Noto font and OFL license")
if(NOT EXISTS "${SHOWCASE_PACK_DIR}/manifest.json")
  message(FATAL_ERROR "Set SHOWCASE_PACK_DIR to the current verified shared pack")
endif()
if(NOT EXISTS "${SHOWCASE_FONT_DIR}/NotoSansSC.ttf" OR NOT EXISTS "${SHOWCASE_FONT_DIR}/OFL.txt")
  message(FATAL_ERROR "Set SHOWCASE_FONT_DIR to the verified bundled font directory")
endif()
find_package(WeappContainerDesktop CONFIG REQUIRED)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_executable(container_showcase MACOSX_BUNDLE macos.mm)
  set_source_files_properties(macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(container_showcase PRIVATE "-framework AppKit")
  target_link_options(container_showcase PRIVATE "$<$<CONFIG:Release>:-Wl,-S,-x>")
  set_target_properties(container_showcase PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER dev.weapp.container.showcase.native)
elseif(WIN32)
  add_executable(container_showcase windows.cpp)
  target_compile_definitions(container_showcase PRIVATE NOMINMAX _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
  target_link_libraries(container_showcase PRIVATE user32 shell32)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
  add_executable(container_showcase linux.cpp)
  target_link_libraries(container_showcase PRIVATE PkgConfig::GTK)
endif()
target_include_directories(container_showcase PRIVATE third_party)
target_link_libraries(container_showcase PRIVATE WeappContainer::Desktop)
if(WIN32)
  weapp_container_copy_runtime(container_showcase)
endif()
if(APPLE)
  set(SHOWCASE_PACK_DESTINATION "$<TARGET_BUNDLE_CONTENT_DIR:container_showcase>/Resources/portable-pack")
  set(SHOWCASE_FONT_DESTINATION "$<TARGET_BUNDLE_CONTENT_DIR:container_showcase>/Resources/fonts")
else()
  set(SHOWCASE_PACK_DESTINATION "$<TARGET_FILE_DIR:container_showcase>/portable-pack")
  set(SHOWCASE_FONT_DESTINATION "$<TARGET_FILE_DIR:container_showcase>/fonts")
endif()
add_custom_command(TARGET container_showcase POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${SHOWCASE_PACK_DIR}" "${SHOWCASE_PACK_DESTINATION}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${SHOWCASE_FONT_DESTINATION}"
  COMMAND ${CMAKE_COMMAND} -E copy "${SHOWCASE_FONT_DIR}/NotoSansSC.ttf" "${SHOWCASE_FONT_DESTINATION}/NotoSansSC.ttf"
  COMMAND ${CMAKE_COMMAND} -E copy "${SHOWCASE_FONT_DIR}/OFL.txt" "${SHOWCASE_FONT_DESTINATION}/OFL.txt")
